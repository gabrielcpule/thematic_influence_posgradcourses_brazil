# ── Extrair autores — somente DISCENTES ───────────────────────────────────────
# Explode a coluna 'autores' mantendo colunas de produção no contexto
_prod_meta_cols = ["nomeSubTipoProducao", "siglaIes", "nomePrograma", "anoBase"]
_available_meta = [c for c in _prod_meta_cols if c in df_resultado.columns]

df_autores_exploded = df_resultado.copy()
for col in _available_meta:
    df_autores_exploded[f"_meta_{col}"] = df_resultado[col]
df_autores_exploded = df_autores_exploded.explode("autores")

df_autores_norm = pd.json_normalize(df_autores_exploded["autores"].dropna())

# Alinhar índice com df_autores_exploded para recuperar metadados
_aligned_idx = df_autores_exploded.dropna(subset=["autores"]).index
for col in _available_meta:
    _meta_col = f"_meta_{col}"
    if _meta_col in df_autores_exploded.columns:
        df_autores_norm[col] = df_autores_exploded.loc[_aligned_idx, _meta_col].values[:len(df_autores_norm)]

# Filtrar apenas Discentes
if "tipoVinculo" in df_autores_norm.columns:
    df_discentes = df_autores_norm[df_autores_norm["tipoVinculo"] == "Discente"].copy()
    print(f"Discentes: {len(df_discentes):,}  |  "
          f"Docentes/outros excluídos: {len(df_autores_norm) - len(df_discentes):,}")
else:
    print("⚠️  Campo 'tipoVinculo' não encontrado — usando todos os autores")
    df_discentes = df_autores_norm.copy()

# Metadata primária por autor (sub-tipo e IES mais frequentes)
_agg_dict = {}
if "nomeSubTipoProducao" in df_discentes.columns:
    _agg_dict["sub_tipo"] = ("nomeSubTipoProducao",
                             lambda s: s.mode().iat[0] if not s.mode().empty else None)
if "siglaIes" in df_discentes.columns:
    _agg_dict["sigla_ies"] = ("siglaIes",
                              lambda s: s.mode().iat[0] if not s.mode().empty else None)
if "nomePrograma" in df_discentes.columns:
    _agg_dict["nome_programa"] = ("nomePrograma",
                                  lambda s: s.mode().iat[0] if not s.mode().empty else None)

if _agg_dict:
    df_discentes_meta = (df_discentes.groupby("nomePessoa")
                         .agg(**_agg_dict)
                         .reset_index())
else:
    df_discentes_meta = df_discentes[["nomePessoa"]].drop_duplicates().reset_index(drop=True)

pesquisadores_unicos = df_discentes["nomePessoa"].unique()
df_pesquisadores = (pd.DataFrame(pesquisadores_unicos, columns=["nomePessoa"])
                    .sort_values("nomePessoa").reset_index(drop=True)
                    .merge(df_discentes_meta, on="nomePessoa", how="left"))

print(f"Discentes únicos (CAPES): {len(df_pesquisadores):,}")
if "sub_tipo" in df_pesquisadores.columns:
    print(f"  Por tipo de produção:\n{df_pesquisadores['sub_tipo'].value_counts().to_string()}")
if "sigla_ies" in df_pesquisadores.columns:
    print(f"  Instituições únicas: {df_pesquisadores['sigla_ies'].nunique()}")
